def FREQ(L):
    D = {}
    for i in L:
        if i in D:
            D[i] += 1
        else:
            D[i] = 1
    return D

print(FREQ([4, 2, 4, 7, 2, 4]))

import pickle

def addrec():
    f = open("STUDENT.DAT", "ab")
    adm = int(input("Admission No: "))
    name = input("Name: ")
    per = float(input("Percent: "))
    pickle.dump([adm, name, per], f)
    f.close()

def countrec():
    count = 0
    f = open("STUDENT.DAT", "rb")
    try:
        while True:
            rec = pickle.load(f)
            if rec[2] > 75:
                print(rec)
                count += 1
    except EOFError:
        f.close()
    return count

def ShowLowStock():
    count = 0
    f = open("LIBRARY.CSV", "r")
    r = csv.reader(f)
    for rec in r:
        if int(rec[3]) < 5:

            count += 1
    f.close()
    print("Total books with low stock:", count)
