def COUNT_CHARS():
    f = open("DATA.TXT", "r")
    data = f.read()
    u = l = d = 0
    for ch in data:
        if ch.isupper():
            u += 1
        elif ch.islower():
            l += 1
        elif ch.isdigit():
            d += 1
    print("Uppercase:", u, "Lowercase:", l, "Digits:", d)
    f.close()

def FREQ(L):
    D = {}
    for i in L:
        if i in D:
            D[i] += 1
        else:
            D[i] = 1
    return D

print(FREQ([4, 2, 4, 7, 2, 4]))

def addrec():                                   # (b) 2 marks

    pickle.dump([adm, name, per], f)

def update_percent(adm):                        # (c) 2 marks
    records = []
    found = False
    f = open("STUDENT.DAT", "rb")
    try:
        while True:
            rec = pickle.load(f)
            if rec[0] == adm:
                rec[2] += 5
                found = True
            records.append(rec)
    except EOFError:
        f.close()
    if found:
        f = open("STUDENT.DAT", "wb")
        for rec in records:
            pickle.dump(rec, f)
        f.close()
    else:
        print("Record not found")
